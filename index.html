Server za računar (C:\SERVER\server.py na portu 3000) Pošto već koristiš port 3000 (vjerovatno preko Node.js http-server ili Live Server ili Pythona), umjesto običnog statičkog servera pokreni ovu skriptu u C:\SERVER\server.py (ugasi prethodni server na portu 3000 prije pokretanja ovoga). Ona radi sve na jednom mjestu: • Kad na telefonu ili računaru otvoriš http://10.53.98.10:3000 -> otvara tvoj index.html. • Kad klikneš Pošalji inventuru na C:\SERVER -> snima podatke u C:\SERVER\inventura.txt. • Kad na računaru (ili telefonu) otvoriš http://10.53.98 -> odmah prikazuje ili preuzima taj fajl. python from http.server import HTTPServer, BaseHTTPRequestHandler import os FOLDER = r"C:\SERVER" os.makedirs(FOLDER, exist_ok=True) class ServerHandler(BaseHTTPRequestHandler): def _cors(self): self.send_header("Access-Control-Allow-Origin", "*") self.send_header("Access-Control-Allow-Methods", "GET, POST, OPTIONS") self.send_header("Access-Control-Allow-Headers", "Content-Type") def do_OPTIONS(self): self.send_response(200) self._cors() self.end_headers() def do_GET(self): # Otvaranje same aplikacije (index.html) if self.path == "/" or self.path == "/index.html": putanja = os.path.join(FOLDER, "index.html") if os.path.exists(putanja): self.send_response(200) self._cors() self.send_header("Content-Type", "text/html; charset=utf-8") self.end_headers() with open(putanja, "rb") as f: self.wfile.write(f.read()) else: self.send_error(404, "Nema index.html u C:\\SERVER") # Pregled i preuzimanje inventura.txt sa bilo kog uredjaja elif self.path == "/inventura.txt": putanja = os.path.join(FOLDER, "inventura.txt") if os.path.exists(putanja): self.send_response(200) self._cors() self.send_header("Content-Type", "text/plain; charset=utf-8") self.end_headers() with open(putanja, "rb") as f: self.wfile.write(f.read()) else: self.send_response(404) self._cors() self.send_header("Content-Type", "text/plain; charset=utf-8") self.end_headers() self.wfile.write("Fajl inventura.txt jos uvijek ne postoji.".encode("utf-8")) else: self.send_error(404) def do_POST(self): if self.path == "/snimi": duzina = int(self.headers.get("Content-Length", 0)) podaci = self.rfile.read(duzina).decode("utf-8") putanja = os.path.join(FOLDER, "inventura.txt") with open(putanja, "w", encoding="utf-8") as f: f.write(podaci) print(f"[OK] Azuriran fajl: {putanja}") self.send_response(200) self._cors() self.send_header("Content-Type", "text/plain; charset=utf-8") self.end_headers() self.wfile.write("Uspjesno snimljeno u C:\\SERVER\\inventura.txt!".encode("utf-8")) print("Server radi na http://10.53.98.10:3000 ...") HTTPServer(("0.0.0.0", 3000), ServerHandler).serve_forever()
